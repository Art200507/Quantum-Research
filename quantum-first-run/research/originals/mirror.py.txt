"""Mirror circuit ("beyond classical") demo.

Build a random layered unitary U (mixed single-qubit rotations + CX/ECR
entanglers), append U.inverse(), and measure. Algebraically U @ U^-1 == I,
so the *ideal* (noiseless) output is the all-zero bitstring with probability
1, for every N. Any weight that leaks onto other bitstrings on real/noisy
hardware is a direct, easy-to-interpret measure of accumulated gate error --
this is the "mirror circuit" / "loschmidt echo" benchmarking technique used
to characterize real devices without needing classical simulation to know
the right answer.

That last point is exactly why this is a good "beyond classical" demo: we
know the correct answer (all zeros) without ever having to classically
compute it, which means we can run it at sizes where classical simulation
is impossible and still grade the hardware's output.

WHY N >= ~50 CAN'T BE STATEVECTOR-SIMULATED
--------------------------------------------
A statevector simulator tracks one complex amplitude per computational basis
state, i.e. 2**N complex numbers, regardless of what the circuit computes:

  N=20: 2**20   ~= 1.0e6   amplitudes  -> ~16 MB   as complex128  (trivial)
  N=25: 2**25   ~= 3.4e7   amplitudes  -> ~0.5 GB   as complex128 (our local cap)
  N=50: 2**50   ~= 1.1e15  amplitudes  -> ~18 PB    as complex128 (no machine has this RAM)
  N=100: 2**100 ~= 1.3e30  amplitudes  -> ~2.0e16 TB as complex128 (physically absurd)

This cost is exponential in N and is completely independent of the fact that
the *answer* happens to be the simple all-zero state -- a statevector
simulator has to track the full superposition at every intermediate layer,
including right after U and before U^-1 cancels it back out, where the state
is maximally scrambled. So even though we know the final answer in advance,
we cannot verify it classically at N=50/100 by simulating the state. Real
QPU time is the only way to get data at those sizes, which is the whole
point of the "beyond classical" framing.

NOISY (FAKETORINO) LOCAL SIMULATION HAS AN EVEN LOWER CEILING
--------------------------------------------------------------
Simulating *with* a realistic noise model is far more expensive than the
noiseless case above: Aer has to re-run the full circuit once per shot
(a separate random noise "trajectory" each time), rather than evolving the
state once and sampling from it many times. Measured on this machine at
optimization_level=3 against FakeTorino, 1000-shot wall time scales roughly
as N=5 -> ~5s, N=10 -> ~10s, N=14 -> ~64s -- so this cost wall arrives
*long* before the memory wall above. Dry-run therefore only executes the
noisy backend for N <= NOISY_LOCAL_CAP; larger sizes are transpiled (for a
realistic depth/gate-count preview) but not executed locally -- consistent
with the fact that --real on actual hardware is the only way to get
meaningful data at N=20 and above anyway.
"""
from __future__ import annotations

import math
import random

from qiskit import QuantumCircuit

from common import Submission

# Largest N for the noiseless Aer correctness check (confirms U;U^-1 == I).
# 2**25 amplitudes (~0.5 GB) is comfortable on a laptop; 2**30+ starts to hurt.
STATEVECTOR_CAP = 25

# Largest N we'll actually execute against the noisy FakeTorino model in
# dry-run. Much lower than STATEVECTOR_CAP -- see the module docstring for
# measured timings showing why Monte Carlo noise trajectories, not memory,
# are the binding constraint here.
NOISY_LOCAL_CAP = 10

# Fixed circuit depth (independent of N) so QPU time stays bounded and
# predictable across sizes -- a deliberate choice given the 10-min/28-day
# runtime budget. A "real" beyond-classical benchmark would scale depth with
# N for maximal entanglement; we trade some of that rigor for cost control.
LAYERS = 8


def _build_mirror_unitary(n_qubits: int, layers: int, rng: random.Random) -> QuantumCircuit:
    qc = QuantumCircuit(n_qubits)
    for layer in range(layers):
        for q in range(n_qubits):
            gate = rng.choice(["sx", "rz", "ry", "rx"])
            if gate == "sx":
                qc.sx(q)
            else:
                theta = rng.uniform(0, 2 * math.pi)  # non-Clifford for rz/ry/rx
                getattr(qc, gate)(theta, q)
        offset = layer % 2  # brick-wall pattern: alternate pairing each layer
        for q in range(offset, n_qubits - 1, 2):
            if rng.random() < 0.5:
                qc.cx(q, q + 1)
            else:
                qc.ecr(q, q + 1)
    return qc


def _build_mirror_circuit(n_qubits: int, layers: int, seed: int) -> QuantumCircuit:
    rng = random.Random(seed)
    u = _build_mirror_unitary(n_qubits, layers, rng)
    # Without a barrier here, the transpiler's gate-cancellation passes notice
    # that U is immediately followed by U^-1 -- an exact palindrome of mutual
    # inverse pairs -- and recursively cancel the ENTIRE circuit down to
    # nothing from the middle out (verified empirically: optimization_level=3
    # collapsed a 100-qubit mirror circuit to depth 1). The barrier blocks
    # commutation/cancellation across the U / U^-1 boundary while still
    # allowing full SABRE-optimized layout and routing on each half.
    full = u.copy()
    full.barrier()
    full.compose(u.inverse(), inplace=True)
    full.measure_all()
    full.name = f"mirror_n{n_qubits}"
    return full


def build(sizes: tuple[int, ...] = (5, 20, 50, 100), layers: int = LAYERS, seed: int = 1234) -> list[Submission]:
    subs = []
    for n in sizes:
        circuit = _build_mirror_circuit(n, layers, seed=seed + n)
        subs.append(
            Submission(
                label=f"mirror_n{n}",
                circuit=circuit,
                meta={
                    "n_qubits": n,
                    "simulate_locally": n <= STATEVECTOR_CAP,  # noiseless ideal check
                    "noisy_locally": n <= NOISY_LOCAL_CAP,  # actual FakeTorino execution
                },
            )
        )
    return subs


def analyze(
    counts_list: list[dict[str, int] | None],
    submissions: list[Submission],
    ideal_counts_map: dict[str, dict[str, int]],
    out_dir: str,
) -> dict:
    results = {}
    ns, fidelities, hammings = [], [], []
    skipped_ns = []

    print("\n--- Mirror circuit (beyond-classical sanity) ---")
    for sub, counts in zip(submissions, counts_list):
        n = sub.meta["n_qubits"]
        if counts is None:
            skipped_ns.append(n)
            ideal = ideal_counts_map.get(sub.label)
            suffix = ""
            if ideal:
                ideal_shots = sum(ideal.values())
                ideal_frac = ideal.get("0" * n, 0) / ideal_shots
                suffix = f"  [ideal/noiseless check: {ideal_frac:.3f}]"
            print(
                f"N={n}: skipped locally (too costly to simulate noisily, see mirror.py) "
                f"-- real hardware only{suffix}"
            )
            results[sub.label] = {"n_qubits": n, "skipped": True, "ideal_all_zero_fraction": ideal_frac if ideal else None}
            continue

        shots = sum(counts.values())
        zero_bits = "0" * n
        all_zero_frac = counts.get(zero_bits, 0) / shots
        avg_hamming = sum(bs.count("1") * c for bs, c in counts.items()) / shots

        ideal = ideal_counts_map.get(sub.label)
        ideal_frac = None
        if ideal:
            ideal_shots = sum(ideal.values())
            ideal_frac = ideal.get(zero_bits, 0) / ideal_shots

        suffix = f"  [ideal/noiseless check: {ideal_frac:.3f}]" if ideal_frac is not None else ""
        print(f"N={n}: all-zero fraction={all_zero_frac:.4f}  avg Hamming weight={avg_hamming:.3f}{suffix}")

        ns.append(n)
        fidelities.append(all_zero_frac)
        hammings.append(avg_hamming)
        results[sub.label] = {
            "n_qubits": n,
            "skipped": False,
            "all_zero_fraction": all_zero_frac,
            "avg_hamming_weight": avg_hamming,
            "ideal_all_zero_fraction": ideal_frac,
        }

    _plot(ns, fidelities, skipped_ns, out_dir)
    return results


def _plot(ns: list[int], fidelities: list[float], skipped_ns: list[int], out_dir: str) -> None:
    import matplotlib.pyplot as plt

    fig, ax = plt.subplots(figsize=(6, 4))
    if ns:
        ax.plot(ns, fidelities, "o-", label="all-zero fraction (measured)")
    for n in skipped_ns:
        ax.axvline(n, color="gray", linestyle=":", alpha=0.6)
        ax.text(n, 0.02, f"N={n}\nnot locally\nsimulated", ha="center", fontsize=8, color="gray")
    ax.set_xlabel("N (qubits)")
    ax.set_ylabel("fraction of all-zero outcomes (fidelity proxy)")
    ax.set_ylim(-0.05, 1.05)
    ax.set_title("Mirror circuit fidelity vs N")
    if ns:
        ax.legend()

    fig.tight_layout()
    path = f"{out_dir}/mirror_fidelity_vs_n.png"
    fig.savefig(path, dpi=130)
    plt.close(fig)
    print(f"Saved plot: {path}")
